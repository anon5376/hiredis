cmake_minimum_required(VERSION 3.15)
project(timeout_portability C)
set(DISABLE_TESTS ON CACHE BOOL "" FORCE)
set(ENABLE_NUGET OFF CACHE BOOL "" FORCE)
add_subdirectory(.. hiredis)
add_executable(timeout-regression timeout-regression.c)
set_target_properties(timeout-regression PROPERTIES C_STANDARD 99)
target_link_libraries(timeout-regression PRIVATE hiredis)
option(TIMEOUT_TEST_LIBEVENT "Run the libevent deadline regression" OFF)
if(TIMEOUT_TEST_LIBEVENT)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBEVENT REQUIRED IMPORTED_TARGET libevent)
    target_compile_definitions(timeout-regression PRIVATE HIREDIS_TEST_ASYNC=1)
    target_link_libraries(timeout-regression PRIVATE PkgConfig::LIBEVENT)
endif()
enable_testing()
add_test(NAME timeout-regression COMMAND timeout-regression)
